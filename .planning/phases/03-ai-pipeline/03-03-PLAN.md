Example pattern for retry test:
```python
def test_analyze_job_retries_on_bad_json(monkeypatch, base_cv, sample_job_text):
    call_count = []

    def fake_run(cmd, **kwargs):
        call_count.append(1)
        if len(call_count) == 1:
            return _make_proc("not valid json at all")
        return _make_proc(JOB_ANALYSIS_JSON)

    mock_subprocess = types.SimpleNamespace(
        run=fake_run,
        TimeoutExpired=subprocess.TimeoutExpired,
    )
    monkeypatch.setattr(pipeline, "subprocess", mock_subprocess)
    result = analyze_job(base_cv, sample_job_text)
    assert isinstance(result, JobAnalysis)
    assert len(call_count) == 2  # failed once, succeeded on retry
```

Example pattern for timeout test:
```python
def test_analyze_job_raises_on_timeout(monkeypatch, base_cv, sample_job_text):
    def fake_run(cmd, **kwargs):

    mock_subprocess = types.SimpleNamespace(
        run=fake_run,
        TimeoutExpired=subprocess.TimeoutExpired,
    )
    monkeypatch.setattr(pipeline, "subprocess", mock_subprocess)
    with pytest.raises(RuntimeError, match="timed out"):
        analyze_job(base_cv, sample_job_text)
```
